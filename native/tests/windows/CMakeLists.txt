cmake_minimum_required(VERSION 3.18)
project(g1_windows_platform_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(G1_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(ocs2 CONFIG REQUIRED)
add_executable(g1_ad_codegen_smoke ad_codegen_smoke.cpp)
target_compile_definitions(g1_ad_codegen_smoke PRIVATE NOMINMAX _USE_MATH_DEFINES)
target_compile_options(g1_ad_codegen_smoke PRIVATE -O0 -Wa,-mbig-obj)
target_include_directories(g1_ad_codegen_smoke SYSTEM PRIVATE ${ocs2_INCLUDE_DIRS})
enable_testing()
add_test(NAME ad_generate COMMAND g1_ad_codegen_smoke)
set_tests_properties(ad_generate PROPERTIES FIXTURES_SETUP ad_library)
add_test(NAME ad_fresh_reload COMMAND g1_ad_codegen_smoke reload)
set_tests_properties(ad_fresh_reload PROPERTIES FIXTURES_REQUIRED ad_library)
