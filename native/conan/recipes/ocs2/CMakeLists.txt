cmake_minimum_required(VERSION 3.18)
project(ocs2_package LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(G1_OCS2_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/upstream")
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem log log_setup serialization)
find_package(pinocchio CONFIG REQUIRED)
find_package(urdfdom CONFIG REQUIRED)
find_package(hpipm CONFIG REQUIRED)
find_package(blasfeo CONFIG REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
foreach(dir IN LISTS pinocchio_INCLUDE_DIRS)
  if(EXISTS "${dir}/pinocchio/deprecated")
    include_directories(SYSTEM "${dir}/pinocchio/deprecated")
  endif()
endforeach()
add_compile_definitions(BOOST_MPL_LIMIT_LIST_SIZE=30 BOOST_MPL_LIMIT_VECTOR_SIZE=30 BOOST_MPL_CFG_NO_PREPROCESSED_HEADERS BOOST_FUSION_INVOKE_MAX_ARITY=10 BOOST_FUSION_INVOKE_PROCEDURE_MAX_ARITY=10 BOOST_FUSION_INVOKE_FUNCTION_OBJECT_MAX_ARITY=10 BOOST_ALL_DYN_LINK PINOCCHIO_URDFDOM_TYPEDEF_SHARED_PTR PINOCCHIO_URDFDOM_USE_STD_SHARED_PTR)
set(G1_CPPAD_INTERFACE_SOURCE "${G1_OCS2_SOURCE_DIR}/ocs2_core/src/automatic_differentation/CppAdInterface.cpp")
include(cmake/WindowsCompatibility.cmake)
include(cmake/PinocchioCompatibility.cmake)
set(G1_NUMERICAL_LIBRARY_TYPE STATIC)
if(BUILD_SHARED_LIBS)
  set(G1_NUMERICAL_LIBRARY_TYPE SHARED)
endif()
include(Targets.cmake)
install(TARGETS ocs2_core ocs2_oc ocs2_mpc ocs2_qp_solver ocs2_ddp ocs2_robotic_tools ocs2_pinocchio_interface ocs2_centroidal_model hpipm_catkin ocs2_sqp ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_core/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_oc/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_mpc/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_test_tools/ocs2_qp_solver/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_ddp/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_robotic_tools/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_pinocchio/ocs2_pinocchio_interface/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_pinocchio/ocs2_centroidal_model/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_sqp/hpipm_catkin/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_sqp/ocs2_sqp/include/" DESTINATION include)
install(DIRECTORY "${G1_OCS2_SOURCE_DIR}/ocs2_thirdparty/include/" DESTINATION include)
if(WIN32)
  install(DIRECTORY "${G1_WINDOWS_COMPAT}/include/" DESTINATION include)
endif()
