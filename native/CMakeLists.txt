cmake_minimum_required(VERSION 3.18)
project(g1_ocp_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(WIN32)
  set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
option(G1_BUILD_CORE "Build native model, problem and dynamics libraries" ON)
option(G1_BUILD_MATLAB "Build the MATLAB MEX adapter" ON)
if(G1_BUILD_MATLAB AND NOT G1_BUILD_CORE)
  message(FATAL_ERROR "The MATLAB adapter requires G1_BUILD_CORE=ON")
endif()
if(NOT G1_BUILD_CORE)
  return()
endif()
set(G1_CORE_DIR "${PROJECT_SOURCE_DIR}/src/g1_controller")
if(G1_BUILD_MATLAB)
  find_package(Matlab REQUIRED)
endif()
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem log log_setup serialization)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
find_package(pinocchio CONFIG REQUIRED)
find_package(urdfdom CONFIG REQUIRED)
add_compile_definitions(BOOST_MPL_LIMIT_LIST_SIZE=30 BOOST_MPL_LIMIT_VECTOR_SIZE=30 BOOST_MPL_CFG_NO_PREPROCESSED_HEADERS BOOST_FUSION_INVOKE_MAX_ARITY=10 BOOST_FUSION_INVOKE_PROCEDURE_MAX_ARITY=10 BOOST_FUSION_INVOKE_FUNCTION_OBJECT_MAX_ARITY=10 BOOST_ALL_DYN_LINK PINOCCHIO_URDFDOM_TYPEDEF_SHARED_PTR PINOCCHIO_URDFDOM_USE_STD_SHARED_PTR)
include_directories(SYSTEM ${urdfdom_INCLUDE_DIRS})
# Pinocchio 3's compatibility headers remain necessary for the retained OCS2 API.
foreach(dir IN LISTS pinocchio_INCLUDE_DIRS)
  if(EXISTS "${dir}/pinocchio/deprecated")
    include_directories(SYSTEM "${dir}/pinocchio/deprecated")
  endif()
endforeach()
find_package(ocs2 CONFIG REQUIRED)
include_directories(BEFORE SYSTEM ${ocs2_INCLUDE_DIRS})
include(cmake/NumericalLibraries.cmake)
add_subdirectory(src)
if(G1_BUILD_MATLAB)
  matlab_add_mex(NAME ocs2_binding_mex SRC src/ocs2_binding_mex/BindingMex.cpp
    LINK_TO humanoid_wb_mpc)
  target_include_directories(ocs2_binding_mex PRIVATE "${G1_CORE_DIR}" "${PROJECT_SOURCE_DIR}/include")
  if(WIN32)
    target_sources(ocs2_binding_mex PRIVATE src/ocs2_binding_mex/exports.def)
  endif()
  if(G1_BUILD_CENTROIDAL)
    target_link_libraries(ocs2_binding_mex humanoid_centroidal_mpc)
    target_compile_definitions(ocs2_binding_mex PRIVATE G1_HAS_CENTROIDAL=1)
  endif()
  install(TARGETS ocs2_binding_mex LIBRARY DESTINATION lib RUNTIME DESTINATION lib)
  if(UNIX AND NOT APPLE)
    target_link_options(ocs2_binding_mex PRIVATE -Wl,--disable-new-dtags)
    set_target_properties(ocs2_binding_mex PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  matlab_add_mex(NAME g1_inverse_dynamics_mex SRC src/g1_controller_mex/InverseDynamicsMex.cpp
    LINK_TO humanoid_model)
  install(TARGETS g1_inverse_dynamics_mex LIBRARY DESTINATION lib RUNTIME DESTINATION lib)
  if(UNIX AND NOT APPLE)
    target_link_options(g1_inverse_dynamics_mex PRIVATE -Wl,--disable-new-dtags)
    set_target_properties(g1_inverse_dynamics_mex PROPERTIES BUILD_RPATH "${CMAKE_BINARY_DIR}/lib;$ORIGIN" INSTALL_RPATH "$ORIGIN")
  endif()
endif()
include(cmake/VerifyLayers.cmake)
