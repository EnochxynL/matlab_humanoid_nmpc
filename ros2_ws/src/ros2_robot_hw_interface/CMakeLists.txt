cmake_minimum_required(VERSION 3.16)
project(ros2_robot_hw_interface)
set(CMAKE_CXX_STANDARD 20)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(g1_plant_bridge_msgs REQUIRED)
find_package(robot_model REQUIRED)
find_package(urdf REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(absl REQUIRED)
add_library(g1_ros2_hw STATIC src/ros2_robot_hw_interface.cpp)
target_include_directories(g1_ros2_hw PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(g1_ros2_hw rclcpp g1_plant_bridge_msgs robot_model urdf)
target_link_libraries(g1_ros2_hw Eigen3::Eigen absl::flat_hash_map)
install(TARGETS g1_ros2_hw ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_libraries(g1_ros2_hw)
ament_export_dependencies(rclcpp g1_plant_bridge_msgs robot_model urdf Eigen3 absl)
ament_package()
